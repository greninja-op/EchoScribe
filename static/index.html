EchoScribe runs as a local background daemon on http://localhost:8765. Connect your apps directly:

Python (`cli-workflow` integration) 
import httpx

async def transcribe_via_echoscribe(audio_bytes):
    async with httpx.AsyncClient(timeout=10.0) as client:
        files = {"file": ("audio.wav", audio_bytes, "audio/wav")}
        resp = await client.post("http://localhost:8765/api/transcribe", files=files)
        data = resp.json()
        return data.get("transcript") # Cleaned & dictionary-corrected
cURL 
curl -X POST http://localhost:8765/api/transcribe \
